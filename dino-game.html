ctx.strokeStyle = '#ff4757'; ctx.lineWidth = 2; ctx.strokeRect(canvas.width / 2 - 180, 45, 360, 95); ctx.font = '16px "Press Start 2P"'; ctx.fillStyle = '#ff4757'; ctx.textAlign = 'center'; ctx.fillText('G A M E O V E R', canvas.width / 2, 78); ctx.font = '9px "Press Start 2P"'; ctx.fillStyle = '#f4d35e'; ctx.fillText('PRESSIONE ESPAÇO PARA JOGAR', canvas.width / 2, 112); ctx.restore(); } function renderScene() { ctx.save(); // Efeito de Screen Shake (SUPER / Explosões) if (screenShakeTimer > 0) { screenShakeTimer--; const shakeX = (Math.random() - 0.5) * 7; const shakeY = (Math.random() - 0.5) * 7; ctx.translate(shakeX, shakeY); } ctx.clearRect(0, 0, canvas.width, canvas.height); drawBackground(); drawGround(); // Desenha Obstáculos obstacles.forEach(drawObstacle); // Desenha Personagem e Partículas mj.draw(ctx); drawParticles(); // Modo Debug de Hitboxes if (showHitboxes) { drawHitBox(mj.getBox()); obstacles.forEach(obs => { if (!obs.destroyed) { const inset = HITBOX_INSETS[obs.type]; drawHitBox({ x: obs.x + inset.left, y: obs.y + inset.top, width: obs.width - inset.left - inset.right, height: obs.height - inset.top - inset.bottom }); } }); } drawHud(); // Screen Flash quando aciona o SUPER if (screenFlashAlpha > 0) { ctx.fillStyle = `rgba(255, 255, 255, ${screenFlashAlpha})`; ctx.fillRect(0, 0, canvas.width, canvas.height); screenFlashAlpha -= 0.05; } ctx.restore(); } function gameLoop() { if (!gameRunning) return; frameCount++; const settings = difficulties[selectedDifficulty]; // Atualização do SUPER if (superActiveFrames > 0) { superActiveFrames--; } else if (superCharge < SUPER_CHARGE_FRAMES) { superCharge++; } if (frameCount % 10 === 0) { updateSuperButton(); } // Progressão de Velocidade e Pontuação distance += gameSpeed * 0.025; const newScore = Math.floor(distance); if (newScore > score) { score = newScore; if (score > 0 && score % 100 === 0) playSound('score'); } gameSpeed = Math.min(settings.maxSpeed, settings.startSpeed + score * settings.speedFactor); mj.update(); updateParticles(); // Gerenciamento Dinâmico de Obstáculos obstacleDistanceRemaining -= gameSpeed; if (obstacleDistanceRemaining <= 0) { spawnObstacle(); obstacleDistanceRemaining = getNextObstacleGap(); } for (let i = 0; i < obstacles.length; i++) { const obs = obstacles[i]; obs.x -= gameSpeed; if (obs.type === 'airplane') { obs.y = obs.baseY + Math.sin(frameCount * 0.08 + obs.phase) * 3; } // Checagem de Colisão Fair if (!obs.destroyed) { const mjBox = mj.getBox(); const inset = HITBOX_INSETS[obs.type]; const obsBox = { x: obs.x + inset.left, y: obs.y + inset.top, width: obs.width - inset.left - inset.right, height: obs.height - inset.top - inset.bottom }; if (checkCollision(mjBox, obsBox)) { triggerGameOver(); return; } } } // Remove obstáculos fora da tela obstacles = obstacles.filter(obs => obs.x > -80); renderScene(); requestAnimationFrame(gameLoop); } // Inicialização do Estado setDifficulty(selectedDifficulty); renderScene();
